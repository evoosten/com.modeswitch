function formatApplianceKwh(v){
  const n = Number(v);
  if(!Number.isFinite(n) || n <= 0) return '';

  return ` · ${n.toLocaleString(
    ({nl:'nl-NL',en:'en-US',de:'de-DE',fr:'fr-FR',es:'es-ES'}[APP_LANG] || 'en-US'),
    {
      minimumFractionDigits: 2,
      maximumFractionDigits: 3
    }
  )} kWh`;
}
